(* ========================================================================== *)
(* TEST SCOPE SHAPE                                                           *)
(* ========================================================================== *)

(*
a node is one of:
- scope      a directory that mirrors an apiscope directory
- test_file  a test module that names its topic and its kind
- conftest   the fixture surface for one scope
- fixtures   the data surface: documents and golden
e2e is its own scope and mirrors nothing.
*)

tree
    = { entry };

entry
    = scope
    | test_file
    | conftest
    | fixtures
    ;

scope
    = name, { entry };

test_file
    = topic, ".", kind, ".test.py";

conftest
    = "conftest.py";

fixtures
    = "fixtures", { document | golden };

document
    = "documents", { segment };

golden
    = "golden", { segment };

kind
    = "unit" | "component" | "integration" | "e2e";

topic
    = name, { "-", name };

name
    = letter, { letter | digit | "_" };

segment
    = ? one path segment ?;

letter
    = "a".."z" | "A".."Z";
