- apiscope # a reader for network resources
  - health # check that apiscope is installed and working
  - openapi # browse OpenAPI specifications
    - info # show OpenAPI spec metadata
      + [SOURCE] # alias or path to the OpenAPI spec
    - list # list operations from an OpenAPI spec
      + [SOURCE] # alias or path to the OpenAPI spec
      + [--tag]? # filter by tag
      + [--method]? # filter by HTTP method
    - describe # describe a single operation
      + [SOURCE] # alias or path to the OpenAPI spec
      + [PATH] # operation path
      + [METHOD] # HTTP method
      + [--request]? # show only request fields, omit responses
    - spec # manage OpenAPI spec aliases
      + [-g]? # edit global config instead of project config
      - add # register a new alias for an OpenAPI spec
        + [ALIAS] # alias name
        + [SOURCE] # path or URL to the OpenAPI spec
      - remove # remove a registered alias
        + [ALIAS] # alias name
      - list # list all registered aliases
